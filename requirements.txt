# ============================================================================
# 级联 Pipeline（路线 A）Python 依赖清单
# ============================================================================
# 覆盖链路：VAD → ASR(SenseVoice) → RAG(bge+FAISS) → LLM(Qwen3-14B/vLLM) → TTS(VoxCPM)
#
# 【本清单的来源】
#   全部版本用 `==` 精确锁定，取自一套**实测可跑通**的环境：
#     Python        3.11.17            ← 推荐基线
#     torch         2.13.0+cu130        （CUDA 13.0 轮子，Ampere/3090Ti 可用）
#     NVIDIA 驱动   需支持 CUDA 13
#   不是"大概能装"的清单，而是"与验证过的环境逐版本一致"的清单。
#
# 【Python 版本】建议用 3.11（上面这套就是 3.11 验证的）。
#   若你坚持用 3.12/3.13：本清单 37 个包**都有 cp312/cp313 或 abi3/纯 py 轮子**
#   （已逐个核验 PyPI），装得上；但 3.12+ 这个组合我没有端到端验证过，
#   属于"可装但未验证"。vllm/nanovllm 这类带 CUDA 扩展的包对版本最敏感。
#   另外注意：**不要用 conda base 环境**装这些依赖，建个独立环境更省事。
#
# 【安装步骤（顺序很重要）】
#   conda create -y -n voiceagent python=3.11 && conda activate voiceagent
#
#   # ① 先装 torch 三件套，并指定 CUDA 轮子索引 —— 必须第一步装！
#   #    为什么必须先装：vllm 会重新解析 torch 依赖，若此时 torch 还没装好，
#   #    可能被换成另一个构建版本，导致 CUDA 不匹配。
#   #    为什么带上 torchvision：vllm==0.31.0 硬依赖 torchvision==0.28.0，
#   #    一起装才能保证三者都是 +cu130 构建。
#   pip install torch==2.13.0 torchaudio==2.11.0 torchvision==0.28.0 \
#       --index-url https://download.pytorch.org/whl/cu130
#
#   # ② 再装本清单其余部分
#   #    此时 torch 三件套已满足，pip 不会重装它们
#   pip install -r requirements.txt
#
#   # ③ 装 TTS 引擎（两步，flash-attn 要编译，见文末详细说明）
#   #   export CUDA_HOME=/usr/local/cuda && export PATH=$CUDA_HOME/bin:$PATH
#   #   pip install flash-attn==2.8.3.post1 --no-build-isolation
#   #   pip install nano-vllm-voxcpm==2.0.4
#
#   # ④ 自检（会逐个 import 并报告缺什么）
#   bash start_all.sh
#   或： python -c "import torch,vllm,funasr,sentence_transformers,nest_asyncio; \
#                   print(torch.__version__, torch.cuda.is_available())"
#
# 【② 的索引说明】上面 ① 用的是 --index-url（只认 pytorch 索引）。已核验该索引
#   完整托管 torch 2.13.0 的全部依赖（filelock/sympy/networkx/jinja2/fsspec/
#   setuptools/triton 以及 cuda-toolkit、cuda-bindings、nvidia-cudnn-cu13、
#   nvidia-nccl-cu13 等 CUDA 包），因此不会出现"依赖找不到"。
#   若该索引在你网络下不可达，改用 --extra-index-url（保留 PyPI 作主索引）：
#       pip install torch==2.13.0 torchaudio==2.11.0 torchvision==0.28.0 \
#           --extra-index-url https://download.pytorch.org/whl/cu130
#
# 【注意】
#   · 本文件**不含** TTS 引擎 nanovllm_voxcpm —— 它不在 PyPI 上，
#     必须从源码安装，见文件末尾的说明。
#   · 可选的 LiveKit / Smart Turn / 投机ASR 等组件也未列入（未 pin 版本的包
#     会破坏本文件"全部锁定"的性质），见文件末尾的注释块。

# ============================================================================
# ⚠️ TTS 引擎：必须**单独两步**安装（不能放进上面的清单）
# ============================================================================
# engine/tts.py 需要 `from nanovllm_voxcpm import VoxCPM`。
# 它来自 **Nano-vLLM-VoxCPM** 项目 —— 注意**不是** OpenBMB/VoxCPM 那个仓库：
#
#     PyPI 包名 : nano-vllm-voxcpm
#     import 名 : nanovllm_voxcpm
#     源码仓库  : https://github.com/a710128/nanovllm-voxcpm
#
# 为什么不写进上面的清单：它硬依赖 `flash_attn`，而 PyPI 上 flash-attn
# **只有源码包、没有预编译轮子**，会在安装时触发本地编译（30~60 分钟）。
# 放进清单会让 `pip install -r requirements.txt` 整体变慢甚至失败。
#
# ---- 第 1 步：编译安装 flash-attn（先做，且必须用 --no-build-isolation）----
#   export CUDA_HOME=/usr/local/cuda
#   export PATH=$CUDA_HOME/bin:$PATH          # 让 nvcc 可见（本机 nvcc 3.3 在 /usr/local/cuda/bin）
#   pip install flash-attn==2.8.3.post1 --no-build-isolation
#
#   · --no-build-isolation 是关键：不加的话 pip 会在隔离环境里编译，
#     那里看不到你已装好的 torch，必然失败。
#   · 编译吃内存，可加 MAX_JOBS 控制并行度：MAX_JOBS=8 pip install ...
#   · **已知风险**：flash-attn 官方预编译轮子最高只到 torch 2.10，
#     没有任何 cu13 + torch2.13 + cp311 的轮子（已全量核验 4528 个 wheel）。
#     所以本项目这套 torch 2.13 只能源码编译，且**未经验证**是否编译通过。
#     若报 nvcc/gcc 版本错误，可尝试： export NVCC_PREPEND_FLAGS="-allow-unsupported-compiler"
#
# ---- 第 2 步：安装 TTS 引擎（flash-attn 已满足，不会再触发编译）----
#   pip install nano-vllm-voxcpm==2.0.4
#   python -c "import nanovllm_voxcpm; print('TTS OK')"
#
# 装好后 TTS 权重用 models/VoxCPM1.5/（仓库里已有）——那是**模型权重**，
# 与上面这个**推理引擎**是两回事，别再 clone OpenBMB/VoxCPM 了。
#
# 附：nano-vllm-voxcpm 2.0.4 声明的依赖（供参考）
#     torch!=2.6.*,>=2.5.0 / torchaudio / triton>=3.0.0 / transformers>=4.51.0
#     xxhash / tqdm / numpy / librosa / pydantic / soundfile>=0.13.1
#     flash_attn（Linux）/ torchcodec
#     → 这些在本文件第 1~11 节里基本都已锁定或会被 pip 解析，无冲突
